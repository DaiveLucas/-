# 壁纸网站完整代码 v3.0

## 一、项目信息

**技术栈**：
- Framework: Next.js 16 (App Router)
- Core: React 19
- Language: TypeScript 5
- UI 组件: shadcn/ui
- Styling: Tailwind CSS 4

**视觉风格**：画廊风（暖灰白背景 #FAFAF7）

**功能列表**：
- 首页：AI搜索 + 分类筛选 + 瀑布流壁纸网格
- 详情页：大图展示 + 收藏/下载/分享 + 相似推荐
- 收藏页：收藏列表 + 批量下载
- 下载历史：下载记录 + 清空历史
- 左侧导航栏：首页/随机/收藏/下载历史/刷新/深色模式
- 响应式：移动端底部Tab栏

```typescript
export function getPopularWallpapers(limit?: number): Wallpaper[] {
  const sorted = [...wallpapers].sort((a, b) => (b.downloads || 0) - (a.downloads || 0));
  return limit ? sorted.slice(0, limit) : sorted;
}

// 获取最新壁纸（按ID倒序）
export function getLatestWallpapers(limit?: number): Wallpaper[] {
  const sorted = [...wallpapers].sort((a, b) => parseInt(b.id) - parseInt(a.id));
  return limit ? sorted.slice(0, limit) : sorted;
}

// 格式化分辨率
export function formatResolution(resolution: Resolution): string {
  return `${resolution.width} × ${resolution.height}`;
}

// 格式化数字
export function formatNumber(num: number | undefined): string {
  if (num === undefined) return '0';
  if (num >= 10000) {
    return (num / 10000).toFixed(1) + '万';
  }
  return num.toLocaleString();
}
```

## 五、下载工具 (src/lib/download.ts)

```typescript
import { Wallpaper } from '@/types/wallpaper';

// 下载历史存储
const DOWNLOAD_HISTORY_KEY = 'downloaded-wallpapers';

// 保存下载记录到 localStorage
function saveDownloadHistory(wallpaper: Wallpaper) {
  try {
    const history = JSON.parse(localStorage.getItem(DOWNLOAD_HISTORY_KEY) || '[]');
    const newRecord = {
      wallpaper,
      downloadedAt: new Date().toISOString(),
    };
    // 避免重复，先移除旧记录
    const filtered = history.filter((item: { wallpaper: Wallpaper }) => item.wallpaper.id !== wallpaper.id);
    filtered.unshift(newRecord);
    localStorage.setItem(DOWNLOAD_HISTORY_KEY, JSON.stringify(filtered));
  } catch (e) {
    console.error('保存下载历史失败', e);
  }
}

// 下载单张图片
export async function downloadImage(
  url: string,
  filename: string,
  wallpaper?: Wallpaper
): Promise<void> {
  try {
    const response = await fetch(url);
    if (!response.ok) throw new Error('下载失败');

    const blob = await response.blob();
    const blobUrl = URL.createObjectURL(blob);

    const link = document.createElement('a');
    link.href = blobUrl;
    link.download = filename;
    document.body.appendChild(link);
    link.click();

    // 延迟移除元素和释放 blob URL
    setTimeout(() => {
      if (link.parentNode) {
        document.body.removeChild(link);
      }
      URL.revokeObjectURL(blobUrl);
    }, 1000);

    // 保存下载历史
    if (wallpaper) {
      saveDownloadHistory(wallpaper);
    }
  } catch (error) {
    console.error('下载失败，尝试直接打开', error);
    window.open(url, '_blank');
  }
}

// 批量下载
export async function downloadMultipleImages(
  items: { url: string; filename: string; wallpaper?: Wallpaper }[]
): Promise<void> {
  for (let i = 0; i < items.length; i++) {
    await downloadImage(items[i].url, items[i].filename, items[i].wallpaper);
    if (i < items.length - 1) {
      await new Promise(resolve => setTimeout(resolve, 300));
    }
  }
}
```

## 六、收藏状态管理 (src/hooks/use-favorites.ts)

```typescript
'use client';

import { useState, useEffect, useCallback } from 'react';

const FAVORITES_KEY = 'wallpaper-favorites';

interface FavoriteState {
  [id: string]: boolean;
}

export function useFavorites() {
  const [favorites, setFavorites] = useState<FavoriteState>({});
  const [isLoaded, setIsLoaded] = useState(false);

  // 初始化：从 localStorage 读取
  useEffect(() => {
    const stored = localStorage.getItem(FAVORITES_KEY);
    if (stored) {
      setFavorites(JSON.parse(stored));
    }
    setIsLoaded(true);
  }, []);

  // 同步到 localStorage
  useEffect(() => {
    if (isLoaded) {
      localStorage.setItem(FAVORITES_KEY, JSON.stringify(favorites));
    }
  }, [favorites, isLoaded]);

  const isFavorite = useCallback(
    (id: string) => !!favorites[id],
    [favorites]
  );

  const toggleFavorite = useCallback((id: string) => {
    setFavorites(prev => ({
      ...prev,
      [id]: !prev[id],
    }));
  }, []);

  const removeFavorite = useCallback((id: string) => {
    setFavorites(prev => {
      const next = { ...prev };
      delete next[id];
      return next;
    });
  }, []);

  const favoriteCount = Object.values(favorites).filter(Boolean).length;

  return {
    favorites,
    isLoaded,
    isFavorite,
    toggleFavorite,
    removeFavorite,
    favoriteCount,
  };
}
```

## 七、下载历史管理 (src/hooks/use-download-history.ts)

```typescript
'use client';

import { useState, useEffect, useCallback } from 'react';
import { Wallpaper } from '@/types/wallpaper';

const DOWNLOAD_HISTORY_KEY = 'downloaded-wallpapers';

interface DownloadRecord {
  wallpaper: Wallpaper;
  downloadedAt: string;
}

export function useDownloadHistory() {
  const [downloadHistory, setDownloadHistory] = useState<DownloadRecord[]>([]);
  const [isLoaded, setIsLoaded] = useState(false);

  useEffect(() => {
    const stored = localStorage.getItem(DOWNLOAD_HISTORY_KEY);
    if (stored) {
      setDownloadHistory(JSON.parse(stored));
    }
    setIsLoaded(true);
  }, []);

  const removeDownload = useCallback((id: string) => {
    setDownloadHistory(prev => prev.filter(item => item.wallpaper.id !== id));
  }, []);

  const clearHistory = useCallback(() => {
    setDownloadHistory([]);
  }, []);

  useEffect(() => {
    if (isLoaded) {
      localStorage.setItem(DOWNLOAD_HISTORY_KEY, JSON.stringify(downloadHistory));
    }
  }, [downloadHistory, isLoaded]);

  return {
    downloadHistory,
    isLoaded,
    removeDownload,
    clearHistory,
  };
}
```

## 八、侧边栏组件 (src/components/Sidebar.tsx)

```typescript
'use client';

import { useState, useEffect } from 'react';
import Link from 'next/link';
import { usePathname, useRouter } from 'next/navigation';
import {
  Home,
  Shuffle,
  Heart,
  Download,
  RefreshCw,
  Sun,
  Moon,
  Globe,
  MoreHorizontal,
} from 'lucide-react';
import { wallpapers } from '@/lib/wallpaper-data';

// Tooltip 组件（在组件外部定义）
function Tooltip({ text, children }: { text: string; children: React.ReactNode }) {
  return (
    <div className="relative flex items-center justify-center group/tooltip">
      {children}
      <div className="absolute left-full ml-2 px-2 py-1 bg-popover text-popover-foreground text-xs rounded whitespace-nowrap opacity-0 group-hover/tooltip:opacity-100 transition-opacity pointer-events-none z-50 shadow-md">
        {text}
      </div>
    </div>
  );
}

export default function Sidebar() {
  const pathname = usePathname();
  const router = useRouter();
  const [mounted, setMounted] = useState(false);
  const [isDark, setIsDark] = useState(false);
  const [showMoreMenu, setShowMoreMenu] = useState(false);

  useEffect(() => {
    setMounted(true);
    setIsDark(document.documentElement.classList.contains('dark'));
  }, []);

  const isActive = (path: string) => pathname === path;

  const iconClass = (active: boolean) =>
    `w-5 h-5 transition-colors duration-200 ${active ? 'text-primary' : 'text-muted-foreground hover:text-primary'}`;

  const toggleTheme = () => {
    const newDark = !isDark;
    setIsDark(newDark);
    document.documentElement.classList.toggle('dark', newDark);
    localStorage.setItem('theme', newDark ? 'dark' : 'light');
  };

  const handleRandomWallpaper = () => {
    const randomIndex = Math.floor(Math.random() * wallpapers.length);
    router.push(`/wallpaper/${wallpapers[randomIndex].id}`);
  };

  const handleRefresh = () => {
    router.refresh();
  };

  return (
    <>
      {/* 桌面端侧边栏 */}
      <aside className="hidden md:flex fixed left-0 top-0 bottom-0 w-16 bg-background/80 backdrop-blur-md border-r border-border/20 z-40 flex-col py-4">
        {/* 主要图标 */}
        <nav className="flex flex-col items-center space-y-1 flex-1">
          <Tooltip text="首页">
            <Link href="/" className={`p-3 rounded-lg transition-colors ${isActive('/') ? 'bg-muted' : 'hover:bg-muted/50'}`}>
              <Home className={iconClass(isActive('/'))} />
            </Link>
          </Tooltip>

          <Tooltip text="随机壁纸">
            <button onClick={handleRandomWallpaper} className="p-3 rounded-lg hover:bg-muted/50 transition-colors">
              <Shuffle className={iconClass(false)} />
            </button>
          </Tooltip>

          <Tooltip text="收藏夹">
            <Link href="/favorites" className={`p-3 rounded-lg transition-colors ${isActive('/favorites') ? 'bg-muted' : 'hover:bg-muted/50'}`}>
              <Heart className={iconClass(isActive('/favorites'))} />
            </Link>
          </Tooltip>

          <Tooltip text="下载历史">
            <Link href="/downloads" className={`p-3 rounded-lg transition-colors ${isActive('/downloads') ? 'bg-muted' : 'hover:bg-muted/50'}`}>
              <Download className={iconClass(isActive('/downloads'))} />
            </Link>
          </Tooltip>

          <Tooltip text="刷新">
            <button onClick={handleRefresh} className="p-3 rounded-lg hover:bg-muted/50 transition-colors">
              <RefreshCw className={iconClass(false)} />
            </button>
          </Tooltip>
        </nav>

        {/* 分隔线 */}
        <div className="w-8 h-px bg-border/50 mx-auto my-2" />

        {/* 底部图标 */}
        <div className="flex flex-col items-center space-y-1">
          <Tooltip text={isDark ? '浅色模式' : '深色模式'}>
            <button onClick={toggleTheme} className="p-3 rounded-lg hover:bg-muted/50 transition-colors">
              {mounted && (isDark ? <Sun className={iconClass(false)} /> : <Moon className={iconClass(false)} />)}
            </button>
          </Tooltip>

          <Tooltip text="语言切换">
            <button className="p-3 rounded-lg hover:bg-muted/50 transition-colors">
              <Globe className={iconClass(false)} />
            </button>
          </Tooltip>

          <Tooltip text="更多">
            <button onClick={() => setShowMoreMenu(!showMoreMenu)} className="p-3 rounded-lg hover:bg-muted/50 transition-colors">
              <MoreHorizontal className={iconClass(false)} />
            </button>
          </Tooltip>
        </div>
      </aside>

      {/* 移动端底部 Tab 栏 */}
      <nav className="md:hidden fixed bottom-0 left-0 right-0 h-16 bg-background/90 backdrop-blur-md border-t border-border/20 z-40 flex items-center justify-around px-2">
        <Link href="/" className={`flex flex-col items-center justify-center py-1 px-3 rounded-lg ${isActive('/') ? 'text-primary' : 'text-muted-foreground'}`}>
          <Home className="w-5 h-5" />
          <span className="text-[10px] mt-1">首页</span>
        </Link>
        <button onClick={handleRandomWallpaper} className="flex flex-col items-center justify-center py-1 px-3 rounded-lg text-muted-foreground">
          <Shuffle className="w-5 h-5" />
          <span className="text-[10px] mt-1">随机</span>
        </button>
        <Link href="/favorites" className={`flex flex-col items-center justify-center py-1 px-3 rounded-lg ${isActive('/favorites') ? 'text-primary' : 'text-muted-foreground'}`}>
          <Heart className="w-5 h-5" />
          <span className="text-[10px] mt-1">收藏</span>
        </Link>
        <Link href="/downloads" className={`flex flex-col items-center justify-center py-1 px-3 rounded-lg ${isActive('/downloads') ? 'text-primary' : 'text-muted-foreground'}`}>
          <Download className="w-5 h-5" />
          <span className="text-[10px] mt-1">下载</span>
        </Link>
      </nav>
    </>
  );
}
```
