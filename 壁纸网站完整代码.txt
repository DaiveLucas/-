## 一、项目信息

### 技术栈
- **框架**：Next.js 16 (App Router)
- **核心**：React 19
- **语言**：TypeScript 5
- **UI 组件**：shadcn/ui (基于 Radix UI)
- **样式**：Tailwind CSS 4
- **图标**：Lucide Icons

### 视觉风格
- 画廊风格（极简、留白、作品优先）
- Unsplash 风格瀑布流布局
- 图片直接展示，无卡片包裹感

### 功能列表
- ✅ AI 搜索框（匹配标题+标签）
- ✅ 10 分类筛选（热门/最新/动漫/风景/治愈系/赛博朋克/极简/太空/抽象）
- ✅ 瀑布流壁纸展示（3列响应式）
- ✅ 收藏功能（localStorage 持久化）
- ✅ 下载功能（fetch+blob 跨域支持）
- ✅ 下载历史记录
- ✅ 深色模式切换
- ✅ 左侧竖向导航栏（桌面端）
- ✅ 底部 Tab 栏（移动端）

```typescript
// 获取相似壁纸（同分类）
export function getSimilarWallpapers(id: number, limit = 6): Wallpaper[] {
  const wallpaper = getWallpaperById(id);
  if (!wallpaper) return [];
  return wallpapers
    .filter((w) => w.category === wallpaper.category && w.id !== id)
    .slice(0, limit);
}

// 获取热门壁纸（按下载量排序）
export function getPopularWallpapers(limit?: number): Wallpaper[] {
  const sorted = [...wallpapers].sort((a, b) => (b.downloads ?? 0) - (a.downloads ?? 0));
  return limit ? sorted.slice(0, limit) : sorted;
}

// 获取最新壁纸（按 ID 倒序）
export function getLatestWallpapers(limit?: number): Wallpaper[] {
  const sorted = [...wallpapers].sort((a, b) => b.id - a.id);
  return limit ? sorted.slice(0, limit) : sorted;
}

// 随机获取一张壁纸
export function getRandomWallpaper(): Wallpaper {
  const randomIndex = Math.floor(Math.random() * wallpapers.length);
  return wallpapers[randomIndex];
}

// 格式化数字（如下载量）
export function formatNumber(num: number): string {
  if (num >= 10000) {
    return (num / 10000).toFixed(1) + 'w';
  }
  if (num >= 1000) {
    return (num / 1000).toFixed(1) + 'k';
  }
  return num.toString();
}
```

## 五、下载工具

### 文件：src/lib/download.ts

```typescript
import type { Wallpaper, DownloadRecord } from '@/types/wallpaper';

// 下载单张图片（fetch + blob 方式，支持跨域）
export async function downloadImage(
  url: string,
  filename: string,
  wallpaper?: Wallpaper
): Promise<void> {
  try {
    const response = await fetch(url);
    if (!response.ok) throw new Error('下载失败');

    const blob = await response.blob();
    const blobUrl = URL.createObjectURL(blob);

    const link = document.createElement('a');
    link.href = blobUrl;
    link.download = filename;
    link.style.display = 'none';
    document.body.appendChild(link);
    link.click();

    // 延迟移除元素和释放 blob URL
    setTimeout(() => {
      if (link.parentNode) {
        document.body.removeChild(link);
      }
      URL.revokeObjectURL(blobUrl);
    }, 1000);

    // 记录下载历史
    if (wallpaper) {
      saveDownloadHistory(wallpaper);
    }
  } catch (error) {
    console.error('下载失败:', error);
    // fallback: 直接打开图片
    window.open(url, '_blank');
  }
}

// 批量下载图片
export async function downloadMultipleImages(
  items: { url: string; filename: string }[]
): Promise<void> {
  for (let i = 0; i < items.length; i++) {
    const { url, filename } = items[i];
    await downloadImage(url, filename);
    // 间隔避免浏览器拦截
    if (i < items.length - 1) {
      await new Promise((resolve) => setTimeout(resolve, 500));
    }
  }
}

// 保存下载历史到 localStorage
function saveDownloadHistory(wallpaper: Wallpaper): void {
  const historyKey = 'downloaded-wallpapers';
  const stored = localStorage.getItem(historyKey);
  let history: DownloadRecord[] = stored ? JSON.parse(stored) : [];

  // 检查是否已存在（避免重复）
  const exists = history.some((item) => item.id === wallpaper.id);
  if (!exists) {
    const record: DownloadRecord = {
      id: wallpaper.id,
      title: wallpaper.title,
      imageUrl: wallpaper.imageUrl,
      thumbnailUrl: wallpaper.thumbnailUrl,
      category: wallpaper.category,
      downloadedAt: Date.now(),
    };
    history.unshift(record); // 新记录放前面
    // 最多保留 50 条
    if (history.length > 50) {
      history = history.slice(0, 50);
    }
    localStorage.setItem(historyKey, JSON.stringify(history));
  }
}
```

## 六、收藏状态管理

### 文件：src/hooks/use-favorites.ts

```typescript
'use client';

import { useState, useEffect } from 'react';

const FAVORITES_KEY = 'wallpaper-favorites';

type FavoriteState = { [id: string]: boolean };

export function useFavorites() {
  const [favorites, setFavorites] = useState<FavoriteState>({});
  const [isLoaded, setIsLoaded] = useState(false);

  // 从 localStorage 加载
  useEffect(() => {
    const stored = localStorage.getItem(FAVORITES_KEY);
    if (stored) {
      setFavorites(JSON.parse(stored));
    }
    setIsLoaded(true);
  }, []);

  // 保存到 localStorage
  const saveFavorites = (newFavorites: FavoriteState) => {
    setFavorites(newFavorites);
    localStorage.setItem(FAVORITES_KEY, JSON.stringify(newFavorites));
  };

  // 切换收藏状态
  const toggleFavorite = (id: number) => {
    const newFavorites = { ...favorites };
    newFavorites[id] = !newFavorites[id];
    saveFavorites(newFavorites);
  };

  // 添加收藏
  const addFavorite = (id: number) => {
    if (!favorites[id]) {
      const newFavorites = { ...favorites, [id]: true };
      saveFavorites(newFavorites);
    }
  };

  // 移除收藏
  const removeFavorite = (id: number) => {
    const newFavorites = { ...favorites };
    delete newFavorites[id];
    saveFavorites(newFavorites);
  };

  // 检查是否已收藏
  const isFavorite = (id: number): boolean => {
    return !!favorites[id];
  };

  // 获取收藏 ID 列表
  const getFavoriteIds = (): number[] => {
    return Object.keys(favorites)
      .filter((key) => favorites[key])
      .map(Number);
  };

  // 获取收藏数量
  const favoriteCount = Object.values(favorites).filter(Boolean).length;

  return {
    favorites,
    isLoaded,
    toggleFavorite,
    addFavorite,
    removeFavorite,
    isFavorite,
    getFavoriteIds,
    favoriteCount,
  };
}
```

## 七、下载历史管理

### 文件：src/hooks/use-download-history.ts

```typescript
'use client';

import { useState, useEffect } from 'react';
import type { DownloadRecord } from '@/types/wallpaper';

const DOWNLOAD_HISTORY_KEY = 'downloaded-wallpapers';

export function useDownloadHistory() {
  const [history, setHistory] = useState<DownloadRecord[]>([]);
  const [isLoaded, setIsLoaded] = useState(false);

  // 从 localStorage 加载
  useEffect(() => {
    const stored = localStorage.getItem(DOWNLOAD_HISTORY_KEY);
    if (stored) {
      setHistory(JSON.parse(stored));
    }
    setIsLoaded(true);
  }, []);

  // 移除一条记录
  const removeRecord = (id: number) => {
    const newHistory = history.filter((item) => item.id !== id);
    setHistory(newHistory);
    localStorage.setItem(DOWNLOAD_HISTORY_KEY, JSON.stringify(newHistory));
  };

  // 清空所有记录
  const clearHistory = () => {
    setHistory([]);
    localStorage.removeItem(DOWNLOAD_HISTORY_KEY);
  };

  return {
    history,
    isLoaded,
    removeRecord,
    clearHistory,
    count: history.length,
  };
}
```

```typescript
    const randomWallpaper = wallpapers[randomIndex];
    router.push(`/wallpaper/${randomWallpaper.id}`);
  };

  // 刷新页面
  const handleRefresh = () => {
    router.refresh();
  };

  return (
    <>
      {/* 桌面端侧边栏 */}
      <aside className="hidden md:flex fixed left-0 top-0 h-full w-16 flex-col items-center py-4 bg-background/80 backdrop-blur-md border-r border-border z-40">
        {/* Logo */}
        <Link href="/" className="mb-6">
          <div className="w-8 h-8 rounded-lg bg-primary flex items-center justify-center">
            <span className="text-primary-foreground font-bold text-sm">W</span>
          </div>
        </Link>

        {/* 主要图标 */}
        <nav className="flex flex-col items-center space-y-1 flex-1">
          <Tooltip text="首页">
            <Link
              href="/"
              className={`p-3 rounded-lg transition-colors ${isActive('/') ? 'bg-muted' : 'hover:bg-muted/50'}`}
            >
              <Home className={iconClass(isActive('/'))} />
            </Link>
          </Tooltip>

          <Tooltip text="随机壁纸">
            <button
              onClick={handleRandomWallpaper}
              className="p-3 rounded-lg hover:bg-muted/50 transition-colors"
            >
              <Shuffle className={iconClass(false)} />
            </button>
          </Tooltip>

          <Tooltip text="收藏夹">
            <Link
              href="/favorites"
              className={`p-3 rounded-lg transition-colors ${isActive('/favorites') ? 'bg-muted' : 'hover:bg-muted/50'}`}
            >
              <Heart className={iconClass(isActive('/favorites'))} />
            </Link>
          </Tooltip>

          <Tooltip text="下载历史">
            <Link
              href="/downloads"
              className={`p-3 rounded-lg transition-colors ${isActive('/downloads') ? 'bg-muted' : 'hover:bg-muted/50'}`}
            >
              <Download className={iconClass(isActive('/downloads'))} />
            </Link>
          </Tooltip>

          <Tooltip text="刷新">
            <button
              onClick={handleRefresh}
              className="p-3 rounded-lg hover:bg-muted/50 transition-colors"
            >
              <RefreshCw className={iconClass(false)} />
            </button>
          </Tooltip>
        </nav>

        {/* 底部图标 */}
        <div className="flex flex-col items-center space-y-1">
          <Tooltip text={isDark ? '浅色模式' : '深色模式'}>
            <button
              onClick={toggleTheme}
              className="p-3 rounded-lg hover:bg-muted/50 transition-colors"
            >
              {mounted && (isDark ? <Sun className={iconClass(false)} /> : <Moon className={iconClass(false)} />)}
            </button>
          </Tooltip>

          <Tooltip text="语言切换">
            <button className="p-3 rounded-lg hover:bg-muted/50 transition-colors">
              <Globe className={iconClass(false)} />
            </button>
          </Tooltip>

          <div className="relative">
            <Tooltip text="更多">
              <button
                onClick={() => setShowMoreMenu(!showMoreMenu)}
                className="p-3 rounded-lg hover:bg-muted/50 transition-colors"
              >
                <MoreHorizontal className={iconClass(false)} />
              </button>
            </Tooltip>

            {/* 更多菜单 */}
            {showMoreMenu && (
              <div className="absolute left-full ml-2 top-0 bg-popover border border-border rounded-lg shadow-lg py-2 min-w-32 z-50">
                <button
                  onClick={() => setShowMoreMenu(false)}
                  className="w-full px-4 py-2 text-left text-sm hover:bg-muted transition-colors"
                >
                  关于
                </button>
              </div>
            )}
          </div>
        </div>
      </aside>

      {/* 移动端底部 Tab 栏 */}
      <nav className="md:hidden fixed bottom-0 left-0 right-0 h-16 bg-background/80 backdrop-blur-md border-t border-border z-40 flex items-center justify-around px-4">
        <Link href="/" className={`flex flex-col items-center gap-1 p-2 ${isActive('/') ? 'text-primary' : 'text-muted-foreground'}`}>
          <Home className="w-5 h-5" />
          <span className="text-xs">首页</span>
        </Link>

        <button onClick={handleRandomWallpaper} className="flex flex-col items-center gap-1 p-2 text-muted-foreground">
          <Shuffle className="w-5 h-5" />
          <span className="text-xs">随机</span>
        </button>

        <Link href="/favorites" className={`flex flex-col items-center gap-1 p-2 ${isActive('/favorites') ? 'text-primary' : 'text-muted-foreground'}`}>
          <Heart className="w-5 h-5" />
          <span className="text-xs">收藏</span>
        </Link>

        <Link href="/downloads" className={`flex flex-col items-center gap-1 p-2 ${isActive('/downloads') ? 'text-primary' : 'text-muted-foreground'}`}>
          <Download className="w-5 h-5" />
          <span className="text-xs">下载</span>
        </Link>
      </nav>
    </>
  );
}
```
